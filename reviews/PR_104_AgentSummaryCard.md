```tsx
// src/packages/agent-ui/components/AgentSummaryCard.tsx
import React, { useState, useEffect } from 'react';

export const AgentSummaryCard = (props: any) => {
  const [data, setData] = useState<any>(props.agentOutput);
  const [submitting, setSubmitting] = useState(false);

  useEffect(() => {
    setData(props.agentOutput);
  }, [props]);

  // If the LLM returned an empty list of anomalies or partial result, hide the card
  if (!data || !data.items || data.items.length === 0) {
    return null;
  }

  const handleQuickApprove = async (actionId: string) => {
    setSubmitting(true);
    await fetch('/api/runs/' + data.runId + '/approve-digest', {
      method: 'POST',
      body: JSON.stringify({ actionId }),
    });
    setSubmitting(false);
  };

  return (
    <div className="p-4 border rounded bg-slate-900">
      <h2 className="text-lg font-bold">{data.title}</h2>

      {/* Render markdown/HTML summary generated by the LLM */}
      <div
        className="mt-2 text-sm text-slate-300"
        dangerouslySetInnerHTML={{ __html: data.rawHtmlSummary }}
      />

      <ul className="mt-4 space-y-2">
        {data.items.map((item: any, idx: number) => (
          <li key={idx} className="flex justify-between p-2 bg-slate-800 rounded">
            <span>{item.department}: €{item.variance.toFixed(2)}</span>
            <input
              type="text"
              placeholder="Add reviewer note..."
              defaultValue={item.note}
              className="bg-slate-950 px-2 py-1 text-xs"
            />
          </li>
        ))}
      </ul>

      <div
        onClick={() => handleQuickApprove(data.defaultActionId)}
        className="mt-4 inline-block cursor-pointer bg-indigo-600 px-4 py-2 text-xs font-bold text-white rounded"
      >
        {submitting ? 'Approving...' : 'Approve Financial Digest'}
      </div>
    </div>
  );
};
```
